  const SOUND_PROFILE='BROWN';
  let audioCtx=null;

  function ensureAudio(){
    if(audioCtx){
      if(audioCtx.state==='suspended'){
        audioCtx.resume().catch(()=>{});
      }
      return audioCtx;
    }

    const AC=
      window.AudioContext ||
      window.webkitAudioContext;

    if(!AC){
      return null;
    }

    try{
      audioCtx=new AC();

      const sampleRate=
        audioCtx.sampleRate;

      const length=
        Math.max(
          1,
          Math.floor(
            sampleRate*.075
          )
        );

      noiseBuffer=
        audioCtx.createBuffer(
          1,
          length,
          sampleRate
        );

      const data=
        noiseBuffer.getChannelData(0);

      /*
        Slightly "oily" mechanical noise:
        correlated random values instead of clean white-noise clicks.
      */
      let last=0;

      for(let i=0;i<length;i++){
        const white=
          Math.random()*2-1;

        last=
          last*.70+
          white*.30;

        data[i]=
          last*.82+
          white*.18;
      }

      return audioCtx;
    }catch{
      audioCtx=null;
      noiseBuffer=null;
      return null;
    }
  }

  function brownParams(direction){

        gain:.11
      }

  function playBrownSwitch(direction){
    if(SOUND_PROFILE!=='BROWN'){
      return;
    }

    const ctx=
      ensureAudio();

    if(
      !ctx ||
      !noiseBuffer
    ){
      return;
    }

    const p=
      brownParams(direction);

    const now=
      ctx.currentTime;

    /*
      Mechanical body: filtered correlated noise.
    */
    const noise=
      ctx.createBufferSource();

    noise.buffer=
      noiseBuffer;

    const band=
      ctx.createBiquadFilter();

    band.type='bandpass';
    band.frequency.setValueAtTime(
      p.band,
      now
    );
    band.Q.setValueAtTime(
      1.15,
      now
    );

    const lowpass=
      ctx.createBiquadFilter();

    lowpass.type='lowpass';
    lowpass.frequency.setValueAtTime(
      2600,
      now
    );

    const noiseGain=
      ctx.createGain();

    noiseGain.gain.setValueAtTime(
      .0001,
      now
    );
    noiseGain.gain.exponentialRampToValueAtTime(
      p.gain,
      now+.004
    );
    noiseGain.gain.exponentialRampToValueAtTime(
      .0001,
      now+.058
    );

    /*
      Low tactile thock. Very short, deliberately non-musical.
    */
    const thock=
      ctx.createOscillator();

    thock.type='triangle';
    thock.frequency.setValueAtTime(
      p.low,
      now
    );
    thock.frequency.exponentialRampToValueAtTime(
      Math.max(

    );

    const thockGain=
      ctx.createGain();

      .0001,

      .055,

    );

      .0001,

    );

    /*
      Tiny tactile bump before the body, like a Brown switch leaf.
    */
    const bump=
      ctx.createOscillator();

    bump.type='square';
    bump.frequency.setValueAtTime(
      p.band*1.45,
      now
    );

    const bumpGain=
      ctx.createGain();

      .020,
      now
    );
    bumpGain.gain.exponentialRampToValueAtTime(
      .0001,
      now+.010
    );

    const master=
      ctx.createGain();

    master.gain.setValueAtTime(
      .72,
      now
    );

    let destination=
      master;

    if(ctx.createStereoPanner){
      const pan=
        ctx.createStereoPanner();

      pan.pan.setValueAtTime(
        p.pan,
        now
      );

      master.connect(pan);
      pan.connect(ctx.destination);
    }else{
      master.connect(
        ctx.destination
      );
    }

    noise.connect(band);
    band.connect(lowpass);
    lowpass.connect(noiseGain);
    noiseGain.connect(destination);

    thock.connect(thockGain);
    thockGain.connect(destination);

    bump.connect(bumpGain);
    bumpGain.connect(destination);

    noise.stop(now+.065);

    thock.start(now);
    thock.stop(now+.060);

    bump.start(now);
    bump.stop(now+.012);
  }
